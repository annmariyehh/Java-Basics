public class Variables {
    public static void main(String[] args) {

        int age = 21;
        String name = "Annmariya";
        double height = 5.4;
        char grade = 'A';
        boolean isStudent = true;

        System.out.println("Name: " + name);

        System.out.println("Grade: " + grade);
        System.out.println("Student: " + isStudent);
    }
}
